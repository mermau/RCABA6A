cmake_minimum_required(VERSION 3.25)
project(RCABA6A VERSION 0.9.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(RCABA6A_BUILD_PLUGIN "Build JUCE VST3, AU and Standalone" ON)
include(CTest)

set(RCABA6A_DSP_FILES
    Source/DSP/BA6AEngine.h
    Source/DSP/BA6AEngine.cpp
    Source/DSP/ControlCalibration.h
)
set(RCABA6A_PLUGIN_FILES
    Source/PluginProcessor.h
    Source/PluginProcessor.cpp
    Source/PluginEditor.h
    Source/PluginEditor.cpp
)
# Embedded artwork: panel without legends (generated by assets/tools/generate_panel.py),
# knob filmstrip (assets/tools/render_knobs.py) and the OFL Barlow Condensed fonts.
set(RCABA6A_ASSETS
    assets/panel_clean.jpg
    assets/knob_strip.png
    assets/fonts/barlow_semibold.ttf
    assets/fonts/barlow_medium.ttf
)
foreach(file IN LISTS RCABA6A_DSP_FILES RCABA6A_PLUGIN_FILES RCABA6A_ASSETS)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${file}")
        message(FATAL_ERROR "Missing required file: ${file}")
    endif()
endforeach()

add_library(BA6A_DSP STATIC ${RCABA6A_DSP_FILES})
target_include_directories(BA6A_DSP PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Source")
if(MSVC)
    target_compile_options(BA6A_DSP PRIVATE /W4)
else()
    target_compile_options(BA6A_DSP PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(BA6A_Tests Tests/EngineTests.cpp)
    target_link_libraries(BA6A_Tests PRIVATE BA6A_DSP)
    add_test(NAME DSP_vs_RCA_documentation COMMAND BA6A_Tests)
endif()

if(RCABA6A_BUILD_PLUGIN)
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.16-security
        GIT_SHALLOW TRUE
        GIT_PROGRESS TRUE
    )
    FetchContent_MakeAvailable(JUCE)
    set(RCABA6A_FORMATS VST3 Standalone)
    if(APPLE)
        list(APPEND RCABA6A_FORMATS AU)
    endif()
    set(RCABA6A_ICON_ARGS)
    if(APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app_icon.png")
        list(APPEND RCABA6A_ICON_ARGS ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/assets/app_icon.png")
    endif()
    # Collateral Effects brand (v0.9.1). The manufacturer code changed from
    # "Iaau": hosts list the plugin as a new one under Collateral Effects.
    juce_add_plugin(RCABA6A
        VERSION 0.9.1
        COMPANY_NAME "Collateral Effects"
        COMPANY_WEBSITE "https://github.com/mermau"
        BUNDLE_ID "com.collateraleffects.rcaba6a"
        PLUGIN_MANUFACTURER_CODE Clfx
        PLUGIN_CODE Rb6a
        FORMATS ${RCABA6A_FORMATS}
        PRODUCT_NAME "rcaba6a"
        PLUGIN_NAME "RCA BA-6A"
        ${RCABA6A_ICON_ARGS}
        DESCRIPTION "RCA BA-6A tube limiting amplifier"
        AU_MAIN_TYPE kAudioUnitType_Effect
        VST3_CATEGORIES Fx Dynamics
        IS_SYNTH FALSE
        NEEDS_MIDI_INPUT FALSE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE
    )
    juce_generate_juce_header(RCABA6A)
    juce_add_binary_data(RCABA6A_Assets SOURCES ${RCABA6A_ASSETS})
    target_sources(RCABA6A PRIVATE ${RCABA6A_PLUGIN_FILES})
    target_link_libraries(RCABA6A
        PRIVATE
            BA6A_DSP
            RCABA6A_Assets
            juce::juce_audio_utils
            juce::juce_audio_processors
            juce::juce_dsp
            juce::juce_gui_extra
        PUBLIC
            juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags
    )
    target_compile_definitions(RCABA6A
        PUBLIC
            JUCE_WEB_BROWSER=0
            JUCE_USE_CURL=0
            JUCE_VST3_CAN_REPLACE_VST2=0
            JUCE_DISPLAY_SPLASH_SCREEN=0
    )

    if(BUILD_TESTING)
        # Processor/editor integration test: the plugin sources compiled into a
        # console application (no plugin wrapper needed).
        juce_add_console_app(BA6A_PluginTests PRODUCT_NAME "BA6A_PluginTests")
        juce_generate_juce_header(BA6A_PluginTests)
        target_sources(BA6A_PluginTests PRIVATE Tests/PluginTests.cpp ${RCABA6A_PLUGIN_FILES})
        target_include_directories(BA6A_PluginTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/Source")
        target_compile_definitions(BA6A_PluginTests PRIVATE
            JucePlugin_Name="RCA BA-6A"
            JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
        target_link_libraries(BA6A_PluginTests PRIVATE
            BA6A_DSP
            RCABA6A_Assets
            juce::juce_audio_utils
            juce::juce_audio_processors
            juce::juce_dsp
            juce::juce_gui_extra
            juce::juce_recommended_config_flags
            juce::juce_recommended_warning_flags)
        add_test(NAME Plugin_Integration
                 COMMAND BA6A_PluginTests "${CMAKE_BINARY_DIR}/gui_snapshots")
    endif()
endif()
